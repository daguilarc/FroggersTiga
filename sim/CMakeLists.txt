cmake_minimum_required(VERSION 3.16)
project(froggers_sim_tests CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../src/core")
set(SIM_DIR "${CMAKE_CURRENT_SOURCE_DIR}")

function(add_sim_test name source)
    add_executable(${name} ${source})
    target_include_directories(${name} PRIVATE "${CORE_DIR}" "${SIM_DIR}")
endfunction()

function(add_sim_instrument_test name source)
    add_executable(${name} ${source} OwnedAllocationHooks.cpp)
    target_include_directories(${name} PRIVATE
        "${CORE_DIR}"
        "${SIM_DIR}"
        "${CMAKE_CURRENT_SOURCE_DIR}/../desktop/Source")
    target_compile_definitions(${name} PRIVATE FROGGERS_OWNED_ALLOCATION_INSTRUMENT=1)
endfunction()
